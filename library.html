function handleLoginSubmit(e) { e.preventDefault(); const usernameInput = document.getElementById("loginUsername").value.trim(); const passwordInput = document.getElementById("loginPassword").value.trim(); const errorMsg = document.getElementById("loginErrorMsg"); if (errorMsg) errorMsg.style.display = "none"; // ดึงข้อมูลผู้ใช้จากโหนด users/{username} ใน Firebase db.ref('users/' + usernameInput).once('value', (snapshot) => { const userData = snapshot.val(); // 1. ตรวจสอบว่ามีชื่อผู้ใช้นี้ในฐานข้อมูลหรือไม่ if (!userData) { if (errorMsg) { errorMsg.innerText = "❌ ไม่พบชื่อผู้ใช้นี้ในระบบ กรุณาลงทะเบียนก่อนครับ"; errorMsg.style.display = "block"; } else { alert("❌ ไม่พบชื่อผู้ใช้นี้ในระบบ กรุณาลงทะเบียนก่อนครับ"); } return; } // 2. ตรวจสอบว่ารหัสผ่านตรงกับในฐานข้อมูลหรือไม่ if (userData.password !== passwordInput) { if (errorMsg) { errorMsg.innerText = "❌ รหัสผ่านไม่ถูกต้อง กรุณาลองใหม่อีกครั้ง"; errorMsg.style.display = "block"; } else { alert("❌ รหัสผ่านไม่ถูกต้อง กรุณาลองใหม่อีกครั้ง"); } return; } // 3. ข้อมูลถูกต้องตรงกัน บันทึกสิทธิ์และข้อมูลลงใน LocalStorage const userSession = { username: usernameInput, fullName: userData.fullName || usernameInput, role: userData.role || 'user', // ดึงสิทธิ์แอดมินหรือผู้ใช้ทั่วไปตรงตาม DB avatar: userData.avatar || `https://api.dicebear.com/7.x/bottts/svg?seed=${usernameInput}` }; localStorage.setItem("ku_user", JSON.stringify(userSession)); alert(`🎉 ลงชื่อเข้าใช้สำเร็จ! ยินดีต้อนรับคุณ ${userSession.fullName}`); location.reload(); }, (error) => { alert("⚠️ เกิดข้อผิดพลาดในการเชื่อมต่อกับฐานข้อมูล"); }); }
